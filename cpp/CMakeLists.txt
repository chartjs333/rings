cmake_minimum_required(VERSION 3.16)
project(rings_fly_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP)
add_executable(rings_fly_cpp main.cpp)
add_executable(rings_graph_test test.cpp)
foreach(target rings_fly_cpp rings_graph_test)
 if(OpenMP_CXX_FOUND)
  target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
 endif()
 if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${target} PRIVATE -ffp-contract=off)
 endif()
endforeach()
enable_testing()
add_test(NAME graph COMMAND rings_graph_test)
